  2. {{ t('Give the KMS container a name') }} ({{ t('type') }} Host (A))
    {{ t('Domain name') }} {{ target }} · {{ t('IP address') }} {{ host if host_is_ip else t('IP of this container') }}

    {{ t('Some versions ask for service and protocol separately: use _vlmcs and _tcp with domain {domain}.', domain=domain) }}
